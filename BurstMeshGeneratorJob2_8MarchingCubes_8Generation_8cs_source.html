167 private static int GetMcCubeIndexAt(int3 pos, FieldData grid, float isoLevel)
168 {
169 int cubeIndex = 0;
170
171 if (grid.DensityAt(pos.x, pos.y, pos.z) > isoLevel)
172 cubeIndex |=
173 1; // the |= operator sets every bit that is set in the right operand to 1 in the left operand also to 1.
174 // so if the left number (cubeIndex) is 0b11001000 and the right one (the mask, lets say 4) is 0b00000100 the result of that operation would be 0b11001100
175 if (grid.DensityAt(pos.x + 1, pos.y, pos.z) > isoLevel) cubeIndex |= 2;
176 if (grid.DensityAt(pos.x, pos.y + 1, pos.z) > isoLevel) cubeIndex |= 4;
177 if (grid.DensityAt(pos.x + 1, pos.y + 1, pos.z) > isoLevel) cubeIndex |= 8;
178 if (grid.DensityAt(pos.x, pos.y, pos.z + 1) > isoLevel) cubeIndex |= 16;
179 if (grid.DensityAt(pos.x + 1, pos.y, pos.z + 1) > isoLevel) cubeIndex |= 32;
180 if (grid.DensityAt(pos.x, pos.y + 1, pos.z + 1) > isoLevel) cubeIndex |= 64;
181 if (grid.DensityAt(pos.x + 1, pos.y + 1, pos.z + 1) > isoLevel) cubeIndex |= 128;
182
183 return cubeIndex;
184 }
