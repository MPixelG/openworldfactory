271 private static int GetCubeIndexAt(int3 pos, FieldData grid, float isoLevel)
272 {
273 int cubeIndex = 0;
274
275 if (grid.DensityAt(pos.x, pos.y, pos.z) > isoLevel)
276 cubeIndex |=
277 1; // the |= operator sets every bit that is set in the right operand to 1 in the left operand also to 1.
278 // so if the left number (cubeIndex) is 0b11001000 and the right one (the mask, lets say 4) is 0b00000100 the result of that operation would be 0b11001100
279 if (grid.DensityAt(pos.x + 1, pos.y, pos.z) > isoLevel) cubeIndex |= 2;
280 if (grid.DensityAt(pos.x + 1, pos.y + 1, pos.z) > isoLevel) cubeIndex |= 4;
281 if (grid.DensityAt(pos.x, pos.y + 1, pos.z) > isoLevel) cubeIndex |= 8;
282 if (grid.DensityAt(pos.x, pos.y, pos.z + 1) > isoLevel) cubeIndex |= 16;
283 if (grid.DensityAt(pos.x + 1, pos.y, pos.z + 1) > isoLevel) cubeIndex |= 32;
284 if (grid.DensityAt(pos.x + 1, pos.y + 1, pos.z + 1) > isoLevel) cubeIndex |= 64;
285 if (grid.DensityAt(pos.x, pos.y + 1, pos.z + 1) > isoLevel) cubeIndex |= 128;
286
287 return cubeIndex;
288 }

299 // lerps (linear interpolates) between 2 given points based on their density values and the iso level
300 private float3 VertexInterp(float isoLevel, int3 p1, int3 p2, float valP1, float valP2)
301 {
302 float3 p;
303
304 if (math.abs(isoLevel - valP1) < 0.00001)
305 return new float3(p1.x, p1.y, p1.z) * CellSize;
306 if (math.abs(isoLevel - valP2) < 0.00001)
307 return new float3(p2.x, p2.y, p2.z) * CellSize;
308 if (math.abs(valP1 - valP2) < 0.00001)
309 return new float3(p1.x, p1.y, p1.z) * CellSize;
310
311 float mu = (isoLevel - valP1) / (valP2 - valP1);
312 p.x = p1.x + mu * (p2.x - p1.x);
313 p.y = p1.y + mu * (p2.y - p1.y);
314 p.z = p1.z + mu * (p2.z - p1.z);
315
316 return p * CellSize;
317 }
