74 public static void Split(this Octree octree, ulong mortonCode, BurstSamplerSettings settings, bool force=false)
75 {

194 public void BuildNodeAsync(ulong mortonCode, byte maxDepth, ref Octree tree)
195 {
196 byte depth = mortonCode.GetDepth();
197
198 JobHandle sampleJob = DensityFieldBuilder.ScheduleBurstDensityFieldDataBuildInTree(
199 _settings,
200 mortonCode,
201 maxDepth,
202 tree.Min,
203 5,
204 out DensityFieldData sample
205 );
206
207 _pendingJobs.Add(sampleJob);
208
209 sampleJob.Complete();
210
211 float minDensity = float.PositiveInfinity;
212 float maxDensity = float.NegativeInfinity;
213
214 foreach (float densitySample in sample.Densities)
215 {
216 minDensity = math.min(minDensity, densitySample);
217 maxDensity = math.max(maxDensity, densitySample);
218 }
219
220
221 OctreeNodeState state = maxDensity < BurstMeshGenerator.IsoLevel
222 ? OctreeNodeState.Empty
223 : minDensity > BurstMeshGenerator.IsoLevel
224 ? OctreeNodeState.Full
225 : OctreeNodeState.Mixed;
226
227 OctreeNode node = new OctreeNode
228 {
229 MortonCode = mortonCode,
230 State = state,
231 ChildMask = 0
232 };
233
234 int nodeIndex = AddNodeAndGetIndex(ref tree, node);
235 sample.Densities.Dispose();
236
237 if (depth >= maxDepth || state != OctreeNodeState.Mixed) return;
238 byte childMask = 0;
239
240 for (int i = 0; i < 8; i++)
241 {
242 BuildNodeAsync(mortonCode.AppendChild((byte)i), maxDepth, ref tree);
243 childMask |= (byte)(1 << i);
244 }
245
246 node.ChildMask = childMask;
247 tree.Nodes[nodeIndex] = node;
248 }
