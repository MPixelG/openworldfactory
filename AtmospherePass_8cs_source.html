104 if (atmosphere == null || _material == null)

110 _material,

112 atmosphere.planetRadius

114
115
116

120 );

122
123 if (_settings.NeedsOpticalDepthPrecompute)
124 {
125 AddOpticalDepthComputePass(

127 opticalDepthHandle

132
133 TextureHandle starMapHandle =

135 _settings.GetStarMapHandle()

143 starMapHandle

148
149 _settings.SetOpticalDepthTexture(_material);

154 resourceData.activeColorTexture;

163 destinationDesc.clearBuffer = false;
164 destinationDesc.depthBufferBits = 0;
