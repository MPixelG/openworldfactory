96 private void DrawChunks()

98 int maxDepth = _planetManager.Octree.MaxDepth;
99 float3 min = _planetManager.Octree.Min;
100 float3 max = _planetManager.Octree.Max;
101
102 int maxOctreeSize = 1 << maxDepth;
103
104 foreach (ulong chunkMeshesCoord in _chunkMeshes.Keys)
105 {
106 byte depth = chunkMeshesCoord.GetDepth();
107 if (depth < minDrawDepth || depth > maxDrawDepth) continue;
108
109 int logicalNodeSize = 1 << (maxDepth - depth);
110 float3 nodeSize = (max - min)*((float)logicalNodeSize/maxOctreeSize);
111
112 float3 localPos = chunkMeshesCoord.DecodeToCoord();
113 float3 worldPos = min + localPos * nodeSize;
114
115 Mesh mesh = _chunkMeshes[chunkMeshesCoord];
116 if (mesh.vertexCount == 0) continue;
117
118 float3 size = max - min;
119
120
121 int submeshCount = _chunkMeshes[chunkMeshesCoord].subMeshCount;
122 for (int submeshIndex = 0; submeshIndex < submeshCount; submeshIndex++)
123 {
124 Material material = _materialDatabase.GetMaterial((VoxelMaterial) submeshIndex);
125 if (material == null)
126 {
127 continue;
128 }
129 var renderParams = new RenderParams(material)
130 {
131 worldBounds = new Bounds(min + size*0.5f, size)
132 };
133
134 Graphics.RenderMesh(
135 renderParams,
136 _chunkMeshes[chunkMeshesCoord],
137 submeshIndex,
138 Matrix4x4.Translate(new Vector3(worldPos.x, worldPos.y, worldPos.z))
139 );
140 }
141 }
